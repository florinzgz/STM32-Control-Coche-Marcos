# 2. CBMC (Bounded Model Checker)
# ============================================================
sudo apt-get update -qq
sudo apt-get install -y cbmc
# Installed: CBMC 5.95.1 (cbmc-5.95.1)
# Path: /usr/bin/cbmc

# Verify:
cbmc --version
# Output: 5.95.1 (cbmc-5.95.1)

# ============================================================
# 3. Infer (Facebook Static Analyzer)
# ============================================================
cd /tmp
wget -q https://github.com/facebook/infer/releases/download/v1.2.0/infer-linux-x86_64-v1.2.0.tar.xz -O infer.tar.xz
tar xf infer.tar.xz
sudo ln -sf /tmp/infer-linux-x86_64-v1.2.0/bin/infer /usr/local/bin/infer
# Installed: Infer v1.2.0
# Path: /usr/local/bin/infer -> /tmp/infer-linux-x86_64-v1.2.0/bin/infer

# Verify:
infer --version
# Output: Infer version v1.2.0
